option(PARHELION_BENCHMARK_OPENCV "build opencv comparisons" ON)
option(PARHELION_BENCHMARK_VIMAGE "build accelerate vimage comparisons" ${APPLE})

if(PARHELION_BENCHMARK_OPENCV)
    find_package(OpenCV CONFIG REQUIRED COMPONENTS core imgproc
        HINTS "${PROJECT_SOURCE_DIR}/third_party/opencv-build")
endif()
if(PARHELION_BENCHMARK_VIMAGE)
    if(NOT APPLE)
        message(FATAL_ERROR "vimage benchmarks require apple accelerate")
    endif()
    find_library(parhelion_accelerate Accelerate REQUIRED)
endif()

function(parhelion_benchmark group kernel backend)
    string(TOLOWER "${backend}" backend_name)
    string(REPLACE "-" "_" kernel_name "${kernel}")
    if(group STREQUAL "threading" OR group STREQUAL "metal")
        set(kernel_name "${group}_${kernel_name}")
    endif()
    set(target "parhelion_bench_${kernel_name}_${backend_name}")
    add_executable(${target} "${group}/${kernel}/${backend}/main.cpp")
    target_link_libraries(${target} PRIVATE parhelion::parhelion)
    # layout benchmarks include the conversion implementation to time raw kernels.
    if(kernel STREQUAL "AoS-SoA")
        target_compile_options(${target} PRIVATE ${parhelion_arch_options})
    endif()
    if(backend STREQUAL "OpenCV")
        target_include_directories(${target} SYSTEM PRIVATE ${OpenCV_INCLUDE_DIRS})
        target_link_libraries(${target} PRIVATE ${OpenCV_LIBS})
    else()
        target_link_libraries(${target} PRIVATE "${parhelion_accelerate}")
    endif()
endfunction()

foreach(backend IN ITEMS OpenCV vImage)
    if(backend STREQUAL "OpenCV" AND NOT PARHELION_BENCHMARK_OPENCV)
        continue()
    endif()
    if(backend STREQUAL "vImage" AND NOT PARHELION_BENCHMARK_VIMAGE)
        continue()
    endif()
    foreach(kernel IN ITEMS AoS-SoA affine blend brighten copy fill grayscale
            histogram premultiply_alpha scale threshold)
        parhelion_benchmark(parhelion-1 ${kernel} ${backend})
    endforeach()
    if(backend STREQUAL "OpenCV")
        parhelion_benchmark(parhelion-1 minmax ${backend})
    endif()
    foreach(kernel IN ITEMS box_blur gaussian_blur laplacian sharpen sobel)
        parhelion_benchmark(parhelion-2 ${kernel} ${backend})
    endforeach()
endforeach()

add_executable(parhelion_bench_pipeline pipeline/main.cpp pipeline/parhelion.cpp)
target_link_libraries(parhelion_bench_pipeline PRIVATE parhelion::parhelion)
if(PARHELION_BENCHMARK_OPENCV)
    target_sources(parhelion_bench_pipeline PRIVATE pipeline/opencv.cpp)
    target_compile_definitions(parhelion_bench_pipeline PRIVATE PARHELION_PIPELINE_OPENCV)
    target_include_directories(parhelion_bench_pipeline SYSTEM PRIVATE ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(parhelion_bench_pipeline PRIVATE ${OpenCV_LIBS})
endif()
if(PARHELION_BENCHMARK_VIMAGE)
    target_sources(parhelion_bench_pipeline PRIVATE pipeline/vimage.cpp)
    target_compile_definitions(parhelion_bench_pipeline PRIVATE PARHELION_PIPELINE_VIMAGE)
    target_link_libraries(parhelion_bench_pipeline PRIVATE "${parhelion_accelerate}")
endif()

foreach(backend IN ITEMS OpenCV vImage)
    if(backend STREQUAL "OpenCV" AND NOT PARHELION_BENCHMARK_OPENCV)
        continue()
    endif()
    if(backend STREQUAL "vImage" AND NOT PARHELION_BENCHMARK_VIMAGE)
        continue()
    endif()
    foreach(kernel IN ITEMS fill copy brighten scale affine threshold blend grayscale premultiply_alpha)
        parhelion_benchmark(threading ${kernel} ${backend})
        if(CMAKE_SYSTEM_NAME STREQUAL "Darwin")
            parhelion_benchmark(metal ${kernel} ${backend})
        endif()
    endforeach()
endforeach()

    target_link_libraries(parhelion_bench_metal_shared PRIVATE parhelion::parhelion)
endif()
